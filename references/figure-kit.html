Copy the FIGKIT TOKENS CSS and the FIGKIT JS script whole; build figures with the API below, never with hand-placed SVG coordinates.

Colors　Inputs in every form are blue; outputs and probabilities orange; learned transforms green; normalization and residual adds yellow; parameters pink; wires, containers, dimensions and notes gray.

7 A layer stack you can open

Move the pointer up and down over the stack. The plate under it is picked by its resting height, its outline takes the role color, and every plate above it lifts away. Names stay out of the drawing and go to the line under it.

Figure 8Line drawing in the Hairline manner: opaque plates, one silhouette and one crease each, the stroke is the only highlight.

8 API

const f = FK.fig('#host', 1260, 470, 'alt text');   // viewBox width ≤ 1260
f.row(y, x0, [                                         // one row, auto-spaced, auto-arrowed
  {id:'emb', box:'Input\nEmbedding', role:'in', w:'xl', param:'\\mathbf{E}'},
  {id:'a0', add:true},
  {id:'d0', dot:true, gap:25},
  {id:'mha', box:'Multi-Head\nAttention', role:'op', w:'l', param:'W^Q W^K W^V W^O'},
]);
// width tiers: s=33 m=60 ffn=73 l=91 xl=85, or a number
f.skip('d0','a1','top');            // residual: branch dot around to ⊕, top|bottom
f.stack('d0','ln2',{label:'N×', side:'top'});  // repeated-block container
f.pe('a0','top');                    // positional-encoding glyph
f.io('Inputs',{to:'emb', role:'in'});           // input label + same-color arrow
f.io('Output\nProbabilities',{from:'sm', role:'out'});
f.wire([[x1,y1],[x2,y1],[x2,y2]], {arrow:true});  // horizontal/vertical only
f.note('K, V', x, y, 'start');      // gray note
f.math('\\role{out}{a} = softmax(\\role{param}{W}\\role{in}{x})', x, y);  // in-figure formula
f.p('mha.l') → [x,y]   // anchors: .l .r .t .b .c
f.light(['mha'])       // highlight; f.light([]) resets
var z = f.zoom('mha', {x, y, w, h, title:'Multi-Head Attention', closed:false, build:function(z){
  z.row(...); z.io(...); z.zoom('sdpa', {...});   // same API inside; (0,0) = panel body top-left
}});
FK.heat('#host', {tokens:[...], w:[[...],...], mode:'ladder'|'pick', causal:true, ctl:'#ctl', role:'in', qrole:'loss'});
FK.iso('#host', {layers:[{name:'Embedding', role:'in'}, ...], ctl:'#ctl', gap:18, open:34});   // bottom to top
FK.lint()              // run before delivery: overlap, tiny text, out of bounds, label overflow
